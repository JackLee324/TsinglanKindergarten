-- =============================================================================
-- 步骤 0（只读，先看清现状再动手）

-- 0.1 列出所有 super_admin 及其"能否登录"的关键字段
SELECT t.username,
       t.status,
       (t.password_hash IS NOT NULL)              AS has_password,
       COALESCE(m.confirmed, false)               AS mfa_enrolled,
       t.failed_login_attempts,
       t.locked_until
  FROM teachers t
  LEFT JOIN teacher_mfa m ON m.teacher_id = t.id
 WHERE t.roles @> ARRAY['super_admin']::varchar[]
 ORDER BY t.username;

-- 0.2 确认你要用的用户名没被占用（teachers 上唯一约束只有主键，
--      username 的唯一性由应用层保证，所以这里要自己查）
SELECT count(*) AS taken FROM teachers WHERE lower(username) = lower('qls-superadmin');


-- =============================================================================
-- 步骤 1 — 方案 A（推荐）：直接新建一个专用的 super_admin 账号
-- =============================================================================
-- 触发器事实（已核实）：
--   trg_teachers_guard_super_admin       = BEFORE UPDATE OR DELETE  → **不含 INSERT**
--   trg_teachers_protect_last_super_admin = BEFORE UPDATE OR DELETE  → **不含 INSERT**
-- 所以 INSERT 一个 roles 含 super_admin 的行是**被允许的**，且不需要声明任何权限。

BEGIN;

INSERT INTO teachers (username, name, name_en, email, roles, status, password_hash)
VALUES (
  'qls-superadmin',                        -- 登录名，改成你的
  '系统超级管理员',                          -- 姓名：NOT NULL 且无默认值，必填
  'System Super Admin',                    -- 英文名，可空
  'superadmin@example.invalid',            -- 邮箱，可空
  ARRAY['super_admin']::varchar[],          -- 角色
  'active',
  '<粘贴由 bootstrap-super-admin.mjs 第 3.1 节生成的完整 scrypt$... 字符串>'
)
RETURNING id, username, roles, status;

COMMIT;

-- 说明：
--   · 只填 NOT NULL 且无默认值的列（name）；id / _created_at / _updated_at / status
--     等都有默认值，不要手写；
--   · password_hash 必须由应用同款 scrypt 参数生成（SCRYPT_N=16384, R=8, P=1,
--     keylen=32, salt=16 字节，salt 与派生密钥均 base64），格式恰好 6 段：
--         scrypt$16384$8$1$<b64 salt>$<b64 derivedKey>
--     手工拼错会静默走到 verifyPassword 的 return false，表现为"用户名或密码错误"，
--     排查成本极高。生成命令见诊断脚本第 3.1 节。
--   · 连接身份：表 owner（如 qlsadmin）不受 RLS 约束。若用受限角色，
--     teachers 上有 INSERT 策略 rls0004_anon_ins_teachers（角色 anon_）可放行，
--     但仍建议用 owner 或有 INSERT 权限的角色。


-- =============================================================================
-- 步骤 1' — 方案 B（备用）：把已有 principal 账号提升为 super_admin
-- =============================================================================
-- 仅在方案 A 不可行时使用。UPDATE 路径**受 guard 触发器约束**，必须在
-- **同一个事务内**声明 super_admin 权限，否则报 42501 insufficient_privilege。
-- 这是测试夹具使用的同一机制（tests/helpers/reset-fixtures.mjs: declareSuperAdmin）。

BEGIN;

-- 第三个参数 true = is_local，作用域仅本事务。
-- ⚠️ 它和下面的 UPDATE **必须在同一事务**；分开发送会丢标记并被拒。
SELECT set_config('app.rbac_actor_super_admin', 'on', true);

UPDATE teachers
   SET roles         = ARRAY['super_admin']::varchar[],
       password_hash = '<粘贴生成的完整 scrypt$... 字符串>'
 WHERE username = '<已有 principal 的登录名>'
RETURNING id, username, roles;

COMMIT;


-- =============================================================================
-- 步骤 2（只读，验收）— 确认账号已就位
-- =============================================================================
-- 期望：has_password = true、mfa_enrolled = false、status = 'active'。
-- 此时"能登录"但"只能访问 @MfaExempt 路由"——这是设计，MFA 走 API 自助绑定。

SELECT t.username,
       t.status,
       (t.password_hash IS NOT NULL) AS has_password,
       COALESCE(m.confirmed, false)  AS mfa_enrolled
  FROM teachers t
  LEFT JOIN teacher_mfa m ON m.teacher_id = t.id
 WHERE t.roles @> ARRAY['super_admin']::varchar[]
 ORDER BY t.username;


-- =============================================================================
-- 步骤 3（只读，最终验收）— MFA 绑定**之后**再跑
-- =============================================================================
-- 期望：mfa_enrolled = true 且 recovery codes 有剩余（confirm 时返回 10 个）。
-- 这两条同时为真，才说明该账号真正"可登录"。

SELECT t.username,
       t.status,
       (t.password_hash IS NOT NULL) AS has_password,
       COALESCE(m.confirmed, false)  AS mfa_enrolled,
       m.enabled_at,
       (SELECT count(*) FROM mfa_recovery_codes rc
         WHERE rc.teacher_id = t.id AND rc.used_at IS NULL) AS recovery_codes_unused
  FROM teachers t
  LEFT JOIN teacher_mfa m ON m.teacher_id = t.id
 WHERE t.roles @> ARRAY['super_admin']::varchar[]
 ORDER BY t.username;
